// Check for Surplus from Previous Month let surplusNote = ""; if (currentMonthIndex > 0) { const prevObj = appData.schedule[currentMonthIndex - 1]; const prevMonthKey = prevObj.month; const prevBen = prevObj.beneficiary; const prevBill = appData.bills[prevMonthKey] || 0; let prevPaidCount = 0; if (appData.payments[prevMonthKey]) { appData.members.forEach(m => { if (appData.payments[prevMonthKey][m]?.paid) prevPaidCount++; }); } const prevCollected = prevPaidCount * amount; // If collected amount exceeded the actual bill, we have a surplus if (prevBill > 0 && prevCollected > prevBill) { const surplus = prevCollected - prevBill; const prevTotal = amount + surplus; surplusNote = `\n\n💡 Note for @${prevBen}: Since there was a surplus of ₹${surplus.toLocaleString('en-IN')} from ${prevMonthKey}, ${prevBen} should contribute ₹${prevTotal.toLocaleString('en-IN')} this month (₹${amount.toLocaleString('en-IN')} monthly contribution + ₹${surplus.toLocaleString('en-IN')} surplus).`;
